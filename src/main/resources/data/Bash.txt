------
#!/bin/bash
# SSL Expiry Alert Script for Amazon Linux 2

# Domain to check
DOMAIN="secure-prints.com"

# Email address where alerts will be sent
ALERT_EMAIL="info@secure-prints.com"

# Get expiry date from certificate
expiry_date=$(echo | openssl s_client -connect $DOMAIN:443 -servername $DOMAIN 2>/dev/null \
  | openssl x509 -noout -enddate | cut -d= -f2)

# Convert to seconds
expiry_sec=$(date -d "$expiry_date" +%s)
now_sec=$(date +%s)

# Days left until expiration
days_left=$(( (expiry_sec - now_sec) / 86400 ))

# Threshold (change if you want alerts earlier)
THRESHOLD=15

if [ $days_left -le $THRESHOLD ]; then
    echo -e "⚠️ SSL certificate for $DOMAIN will expire in $days_left days.\nExpiry Date: $expiry_date" \
    | mail -s "SSL Expiry Alert: $DOMAIN ($days_left days left)" $ALERT_EMAIL
fi
